
Get-ChildItem -Path C:\ -Filter "log4j-core-*.jar" -Recurse -ErrorAction SilentlyContinue | Select FullName, Name

Get-ChildItem -Path C:\ -Filter "log4j-core-*.jar" -Recurse -ErrorAction SilentlyContinue | ForEach-Object {
    if ($_.Name -match "log4j-core-(.+)\.jar") {
        [PSCustomObject]@{
            Path=$_.FullName
            Version=$matches[1]
        }
    }
}




Get-ChildItem C:\ -Filter package.json -Recurse -ErrorAction SilentlyContinue | ForEach-Object { if (Select-String -Path $_.FullName -Pattern '"next"' -Quiet) { $dir=$_.DirectoryName; Write-Host "`nApplication Path: $dir"; Push-Location $dir; npm list next --depth=0 2>$null; Pop-Location } }
