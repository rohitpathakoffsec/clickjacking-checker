Collect-Evidence-Part2.ps1
Read-only Windows evidence collection for vulnerability validation created by VM11 Team (Rohit).
#>

$OutputRoot="C:\VM_Evidence"
New-Item -ItemType Directory -Force -Path $OutputRoot|Out-Null
$Txt=Join-Path $OutputRoot "Part2_Evidence.txt"
$Csv=Join-Path $OutputRoot "Part2_Evidence.csv"
Remove-Item $Txt,$Csv -ErrorAction SilentlyContinue

$Global:HostName=$env:COMPUTERNAME

function Write-Section($Title){
    $hdr="`r`n==================== $Title ====================`r`nHostName : $Global:HostName`r`n"
    $hdr | Tee-Object -FilePath $Txt -Append
    [pscustomobject]@{HostName=$Global:HostName;Section=$Title;Command="---";Output="---"} |
        Export-Csv $Csv -Append -NoTypeInformation
}

function Run-Command{
    param($Section,$Command,$Script)
    Write-Section $Section
    Write-Host "`n[$Global:HostName] $Command`n" -ForegroundColor Cyan
    $out=& $Script 2>&1 | Out-String
    if([string]::IsNullOrWhiteSpace($out)){ $out="No Output"}
    $out | Tee-Object -FilePath $Txt -Append
    [pscustomobject]@{
        HostName=$Global:HostName
        Section=$Section
        Command=$Command
        Output=$out.Trim()
    } | Export-Csv $Csv -Append -NoTypeInformation
}

Run-Command "Chromium" "Edge/Chrome Version" {
 $paths=@(
 "C:\Program Files (x86)\Microsoft\Edge\Application\msedge.exe",
 "C:\Program Files\Microsoft\Edge\Application\msedge.exe",
 "C:\Program Files\Google\Chrome\Application\chrome.exe")
 foreach($p in $paths){if(Test-Path $p){"$env:COMPUTERNAME`t$p`t$((Get-Item $p).VersionInfo.ProductVersion)"}}
}

Run-Command "Windows Server" "OS Information" {
 $cv=Get-ItemProperty "HKLM:\SOFTWARE\Microsoft\Windows NT\CurrentVersion"
 $ci=Get-ComputerInfo
 [pscustomobject]@{
 HostName=$env:COMPUTERNAME
 OSName="$($ci.WindowsProductName) $($cv.DisplayVersion)"
 OSVersion="$($cv.CurrentMajorVersionNumber).$($cv.CurrentMinorVersionNumber).$($cv.CurrentBuild).$($cv.UBR)"
 SystemType=$ci.CsSystemType}|Format-List
}

Run-Command "BitLocker" "manage-bde -status" {manage-bde -status}
Run-Command "BitLocker" "Get-BitLockerVolume" {Get-BitLockerVolume}
Run-Command "cldflt" "driverquery | findstr cldflt" {cmd /c "driverquery | findstr /i cldflt"}
Run-Command "cldflt" "sc query cldflt" {cmd /c "sc query cldflt"}
Run-Command "OpenSSL" "openssl version" {openssl version}
Run-Command ".NET" "dotnet --info" {dotnet --info}

Write-Host "`nCompleted."
Write-Host "TXT : $Txt"
Write-Host "CSV : $Csv"
